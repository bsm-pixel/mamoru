<script>
(function(){
  var KEY = 'mm_cat_switch';
  var catTop = document.querySelector('.cat-top');
  if (!catTop) return;

  /* ── 도착: 탭 전환으로 온 경우 — 즉시 탭 위치 고정 + 하단 fade-in ── */
  if (sessionStorage.getItem(KEY)) {
    sessionStorage.removeItem(KEY);

    /* 탭 아래 모든 형제 요소 숨김 (상품 영역) */
    var siblings = [];
    var el = catTop.parentElement;
    if (el) {
      var children = el.children;
      var found = false;
      for (var i = 0; i < children.length; i++) {
        if (children[i] === catTop) { found = true; continue; }
        if (found) {
          children[i].style.opacity = '0';
          children[i].style.transition = 'opacity 0.3s ease';
          siblings.push(children[i]);
        }
      }
    }

    /* 탭 위치로 즉시 스크롤 (페이지 로드 최초 시점) */
    function scrollToTab() {
      var tabTop = catTop.getBoundingClientRect().top + window.scrollY;
      window.scrollTo(0, Math.max(0, tabTop - 4));
    }
    scrollToTab();

    /* 상품 로드 완료 후 하단 fade-in */
    function showContent() {
      scrollToTab();
      for (var j = 0; j < siblings.length; j++) {
        siblings[j].style.opacity = '1';
      }
    }

    if (document.readyState === 'complete') {
      setTimeout(showContent, 50);
    } else {
      window.addEventListener('load', function() { setTimeout(showContent, 50); });
    }
  }

  /* ── 출발: 탭 클릭 시 하단만 fade-out → 이동 ── */
  var miniTabs = catTop.querySelector('.cat-mini');
  if (!miniTabs) return;

  miniTabs.addEventListener('click', function(e) {
    var a = e.target.closest('a');
    if (!a || a.classList.contains('active')) return;

    e.preventDefault();
    sessionStorage.setItem(KEY, '1');

    /* 탭 아래 형제 요소만 fade-out */
    var parent = catTop.parentElement;
    if (parent) {
      var children = parent.children;
      var found = false;
      for (var i = 0; i < children.length; i++) {
        if (children[i] === catTop) { found = true; continue; }
        if (found) {
          children[i].style.transition = 'opacity 0.15s ease';
          children[i].style.opacity = '0';
        }
      }
    }

    /* fade-out 완료 후 이동 */
    setTimeout(function() {
      window.location.href = a.getAttribute('href');
    }, 160);
  });
})();
</script>
